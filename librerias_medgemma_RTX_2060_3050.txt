# 2. Actualizar herramientas de instalación
python -m pip install --upgrade pip setuptools wheel

# 3. PyTorch para RTX 2060 / RTX 3050 - CUDA 12.4
pip install torch==2.6.0 torchvision==0.21.0 torchaudio==2.6.0 --index-url https://download.pytorch.org/whl/cu124

# 4. MedGemma / Hugging Face
pip install transformers accelerate sentencepiece safetensors huggingface-hub tokenizers

# 5. Procesamiento de imágenes y datos
pip install pillow numpy pandas matplotlib scikit-learn scipy opencv-python

# 6. Gemini y variables de entorno
pip install google-genai python-dotenv

# 7. Audio
pip install soundfile librosa

# 8. Interfaz y utilidades
pip install gradio tqdm requests protobuf packaging psutil einops

# 9. Jupyter
pip install jupyter notebook ipykernel ipywidgets

# 10. Cuantización - recomendable para MedGemma con VRAM limitada
pip install bitsandbytes

# 11. Registrar kernel
python -m ipykernel install --user --name medgemma_radio --display-name "Python (MedGemma Radiologia)"
